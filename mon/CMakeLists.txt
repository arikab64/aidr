find_program(CARGO cargo REQUIRED)

set(CARGO_TARGET_DIR ${CMAKE_CURRENT_BINARY_DIR}/target)

if(CMAKE_BUILD_TYPE MATCHES "^(Release|RelWithDebInfo|MinSizeRel)$")
    set(CARGO_PROFILE_FLAG --release)
    set(CARGO_PROFILE_DIR release)
else()
    set(CARGO_PROFILE_FLAG "")
    set(CARGO_PROFILE_DIR debug)
endif()

set(MON_BIN ${CARGO_TARGET_DIR}/${CARGO_PROFILE_DIR}/mon)

set(MON_OUTPUT_DIR ${CMAKE_BINARY_DIR}/var/aidr)
set(MON_BIN_STAGED ${MON_OUTPUT_DIR}/mon)

file(GLOB_RECURSE MON_SOURCES CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/src/*.rs)

add_custom_command(
    OUTPUT ${MON_BIN}
    COMMAND ${CARGO} build ${CARGO_PROFILE_FLAG}
            --manifest-path ${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml
            --target-dir ${CARGO_TARGET_DIR}
    DEPENDS ${MON_SOURCES}
            ${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml
            ${CMAKE_SOURCE_DIR}/bpf/mon_files.skel.rs
            ${CMAKE_SOURCE_DIR}/bpf/mon_procs.skel.rs
    COMMENT "CARGO ${MON_BIN}"
    VERBATIM
)

add_custom_command(
    OUTPUT ${MON_BIN_STAGED}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${MON_OUTPUT_DIR}
    COMMAND ${CMAKE_COMMAND} -E copy ${MON_BIN} ${MON_BIN_STAGED}
    DEPENDS ${MON_BIN}
    COMMENT "STAGE ${MON_BIN_STAGED}"
    VERBATIM
)

add_custom_target(mon_bin ALL DEPENDS ${MON_BIN_STAGED})
# Order-only: the skeletons (generated by the bpf_* targets) must exist before cargo runs.
add_dependencies(mon_bin bpf_mon_files bpf_mon_procs)
